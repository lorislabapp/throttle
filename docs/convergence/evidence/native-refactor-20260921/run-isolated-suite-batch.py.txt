from pathlib import Path
import subprocess,time,json,datetime
r=Path(__file__).parent;repo=r.parent/'convergence-recovery-20260920';dest=repo/'docs/convergence/evidence/native-refactor-20260921';label='isolated-suite-batch-'+datetime.datetime.now().strftime('%Y%m%d-%H%M%S');start=time.monotonic();results=[]
suites=['PromptRefinerServiceTests','CockpitSpendViewTests','RemoteSessionsCredentialTests','CloudKitPublisherTests','PeerTerminalBoundaryTests','RemoteTransferJournalTests','ProjectAssistantViewTests','PlanIntegrationViewTests','PlanIntegrationFlowTests']
def save():
 (dest/(label+'.json')).write_text(json.dumps({'elapsed_seconds':round(time.monotonic()-start,1),'results':results,'scope':'all nine suites, 84 expected cases, distinct internal test host'},indent=2))
for suite in suites:
 attempts=0
 while True:
  if time.monotonic()-start>900:save();raise SystemExit('Bounded suite batch timed out')

  if p.returncode==0:time.sleep(10);continue

   subprocess.run(['python3','-B',str(r/'export-qualified-evidence.py'),test_label],check=True)
   break

   attempts+=1
   if attempts<3:time.sleep(10);continue
  raise SystemExit('Test batch stopped after failure or retry limit')
save();print('ALL REMAINING SUITES TERMINAL PASS',flush=True)
